if(WIN32)
    target_compile_definitions(btb_game_enhanced PRIVATE UNICODE _UNICODE)
    # Compile the shared compatibility adapter separately. No ddraw.lib,
    # and no possible fallback to DirectDrawCreateEx in the enhanced target.
    get_target_property(BTB_ENHANCED_PLATFORM_SOURCES btb_original_win32_platform SOURCES)
    add_library(btb_enhanced_win32_platform STATIC ${BTB_ENHANCED_PLATFORM_SOURCES})
    target_include_directories(btb_enhanced_win32_platform PUBLIC ${BTB_ROOT}/include)
    btb_require_cpp26(btb_enhanced_win32_platform PUBLIC)
    target_compile_definitions(btb_enhanced_win32_platform PRIVATE BTB_ENHANCED_NO_DDRAW=1)
    target_link_libraries(btb_enhanced_win32_platform PUBLIC
        btb_recompiled_core btb_bink32_recovered dxguid dsound winmm ole32 gdi32 user32 comdlg32)
    target_link_libraries(btb_game_enhanced PRIVATE
        btb_enhanced_win32_platform user32 gdi32 msimg32 comdlg32 shell32)
    target_sources(btb_game_enhanced PRIVATE ${BTB_ROOT}/enhanced/enhanced_host.cpp ${BTB_ROOT}/enhanced/discord_presence.cpp)
    add_executable(btb_enhanced_win32_packed_tests ${BTB_ROOT}/enhanced/win32_packed_scene_test.cpp)
    btb_require_cpp26(btb_enhanced_win32_packed_tests PRIVATE)
    target_link_libraries(btb_enhanced_win32_packed_tests PRIVATE btb_enhanced_win32_platform)
    target_compile_definitions(btb_enhanced_win32_packed_tests PRIVATE BTB_ENHANCED_NO_DDRAW=1)
    add_test(NAME enhanced_win32_packed_scene COMMAND btb_enhanced_win32_packed_tests)
    set_tests_properties(enhanced_win32_packed_scene PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 20)
    option(BTB_REQUIRE_VULKAN "Require a Vulkan 1.4 SDK when building enhanced" OFF)
    find_package(Vulkan 1.4 QUIET)
    if(Vulkan_FOUND)
        target_sources(btb_game_enhanced PRIVATE ${BTB_ROOT}/enhanced/vulkan_presenter.cpp ${BTB_ROOT}/enhanced/vulkan_sprite_batch.cpp ${BTB_ROOT}/enhanced/native_cursor.cpp)
        target_link_libraries(btb_game_enhanced PRIVATE Vulkan::Vulkan)
        add_executable(btb_enhanced_vulkan_tests ${BTB_ROOT}/enhanced/vulkan_smoke_test.cpp ${BTB_ROOT}/enhanced/vulkan_presenter.cpp ${BTB_ROOT}/enhanced/vulkan_sprite_batch.cpp ${BTB_ROOT}/enhanced/discord_presence.cpp ${BTB_ROOT}/enhanced/native_cursor.cpp)
        btb_require_cpp26(btb_enhanced_vulkan_tests PRIVATE)
        target_include_directories(btb_enhanced_vulkan_tests PRIVATE ${BTB_ROOT}/include)
        target_compile_definitions(btb_enhanced_vulkan_tests PRIVATE UNICODE _UNICODE VK_USE_PLATFORM_WIN32_KHR NOMINMAX WIN32_LEAN_AND_MEAN)
        target_link_libraries(btb_enhanced_vulkan_tests PRIVATE Vulkan::Vulkan user32 gdi32)
        add_executable(btb_enhanced_vulkan_sprite_tests ${BTB_ROOT}/enhanced/vulkan_sprite_batch_test.cpp ${BTB_ROOT}/enhanced/vulkan_sprite_batch.cpp)
        btb_require_cpp26(btb_enhanced_vulkan_sprite_tests PRIVATE)
        target_include_directories(btb_enhanced_vulkan_sprite_tests PRIVATE ${BTB_ROOT}/include)
        target_compile_definitions(btb_enhanced_vulkan_sprite_tests PRIVATE VK_USE_PLATFORM_WIN32_KHR NOMINMAX WIN32_LEAN_AND_MEAN)
        target_link_libraries(btb_enhanced_vulkan_sprite_tests PRIVATE Vulkan::Vulkan user32 gdi32 btb_bink32_recovered)
        add_test(NAME enhanced_vulkan_sprite_differential COMMAND btb_enhanced_vulkan_sprite_tests)
        set_tests_properties(enhanced_vulkan_sprite_differential PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 30)
        add_test(NAME enhanced_vulkan_present_resize COMMAND btb_enhanced_vulkan_tests)
        set_tests_properties(enhanced_vulkan_present_resize PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 20)
        target_compile_definitions(btb_game_enhanced PRIVATE BTB_HAS_VULKAN=1 VK_USE_PLATFORM_WIN32_KHR NOMINMAX WIN32_LEAN_AND_MEAN)
        find_program(BTB_SLANGC slangc HINTS "$ENV{VULKAN_SDK}/Bin")
        option(BTB_REQUIRE_SLANG "Require Slang compiler for enhanced shader development" OFF)
        if(BTB_SLANGC)
            set(BTB_SHADER_OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/shaders")
            file(MAKE_DIRECTORY "${BTB_SHADER_OUTPUT}")
            set(BTB_SPIRV_OUTPUTS "")
            foreach(spec "sprite.slang;spriteVS;vertex" "sprite.slang;spritePS;fragment" "bink_yuv.slang;binkConvert;compute" "sprite_batch.slang;spriteBatch;compute" "bink_opaque.slang;binkOpaque;compute")
                list(GET spec 0 source)
                list(GET spec 1 entry)
                list(GET spec 2 stage)
                set(out "${BTB_SHADER_OUTPUT}/${entry}.spv")
                add_custom_command(OUTPUT "${out}"
                    COMMAND "${BTB_SLANGC}" "${BTB_ROOT}/enhanced/shaders/${source}"
                        -entry "${entry}" -stage "${stage}" -target spirv -profile spirv_1_5 -fvk-use-entrypoint-name -o "${out}"
                    DEPENDS "${BTB_ROOT}/enhanced/shaders/${source}" VERBATIM)
                list(APPEND BTB_SPIRV_OUTPUTS "${out}")
            endforeach()
            add_custom_target(btb_enhanced_shaders ALL DEPENDS ${BTB_SPIRV_OUTPUTS})
            add_dependencies(btb_game_enhanced btb_enhanced_shaders)
            add_dependencies(btb_enhanced_vulkan_tests btb_enhanced_shaders)
            add_dependencies(btb_enhanced_vulkan_sprite_tests btb_enhanced_shaders)
            foreach(target btb_game_enhanced btb_enhanced_vulkan_tests btb_enhanced_vulkan_sprite_tests)
                add_custom_command(TARGET ${target} POST_BUILD
                    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:${target}>/shaders"
                    COMMAND ${CMAKE_COMMAND} -E copy_if_different ${BTB_SPIRV_OUTPUTS} "$<TARGET_FILE_DIR:${target}>/shaders" VERBATIM)
            endforeach()
        elseif(BTB_REQUIRE_SLANG)
            message(FATAL_ERROR "slangc is required for enhanced shader builds")
        endif()
    elseif(BTB_REQUIRE_VULKAN)
        message(FATAL_ERROR "Enhanced game requires the Vulkan 1.4 SDK")
    else()
        message(WARNING "Vulkan 1.4 SDK absent: enhanced target will only support source smoke tests")
    endif()
endif()
