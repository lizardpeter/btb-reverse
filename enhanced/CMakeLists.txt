# Separate opt-in product; no enhanced source is linked into btb_game.
add_executable(btb_game_enhanced WIN32
    ${BTB_ROOT}/runtime/game_main.cpp
    ${BTB_ROOT}/runtime/game_win32.cpp)
btb_require_cpp26(btb_game_enhanced PRIVATE)
target_include_directories(btb_game_enhanced PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(btb_game_enhanced PRIVATE BTB_ENHANCED_BUILD=1)
target_link_libraries(btb_game_enhanced PRIVATE btb_game_session btb_recompiled_core)
if(WIN32)
    target_compile_definitions(btb_game_enhanced PRIVATE UNICODE _UNICODE)
    target_link_libraries(btb_game_enhanced PRIVATE
        btb_original_win32_platform user32 gdi32 msimg32 comdlg32 shell32)
    target_sources(btb_game_enhanced PRIVATE enhanced_host.cpp discord_presence.cpp)
    option(BTB_REQUIRE_VULKAN "Require a Vulkan 1.4 SDK when building enhanced" OFF)
    find_package(Vulkan 1.4 QUIET)
    if(Vulkan_FOUND)
        target_sources(btb_game_enhanced PRIVATE vulkan_presenter.cpp)
        target_link_libraries(btb_game_enhanced PRIVATE Vulkan::Vulkan)
        target_compile_definitions(btb_game_enhanced PRIVATE BTB_HAS_VULKAN=1 VK_USE_PLATFORM_WIN32_KHR NOMINMAX WIN32_LEAN_AND_MEAN)
        find_program(BTB_SLANGC slangc HINTS "$ENV{VULKAN_SDK}/Bin")
        option(BTB_REQUIRE_SLANG "Require Slang compiler for enhanced shader development" OFF)
        if(BTB_SLANGC)
            set(BTB_SHADER_OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/shaders")
            file(MAKE_DIRECTORY "${BTB_SHADER_OUTPUT}")
            set(BTB_SPIRV_OUTPUTS "")
            foreach(spec "sprite.slang;spriteVS;vertex" "sprite.slang;spritePS;fragment" "bink_yuv.slang;binkConvert;compute")
                list(GET spec 0 source)
                list(GET spec 1 entry)
                list(GET spec 2 stage)
                set(out "${BTB_SHADER_OUTPUT}/${entry}.spv")
                add_custom_command(OUTPUT "${out}"
                    COMMAND "${BTB_SLANGC}" "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${source}"
                        -entry "${entry}" -stage "${stage}" -target spirv -profile spirv_1_5 -o "${out}"
                    DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${source}" VERBATIM)
                list(APPEND BTB_SPIRV_OUTPUTS "${out}")
            endforeach()
            add_custom_target(btb_enhanced_shaders ALL DEPENDS ${BTB_SPIRV_OUTPUTS})
            add_dependencies(btb_game_enhanced btb_enhanced_shaders)
        elseif(BTB_REQUIRE_SLANG)
            message(FATAL_ERROR "slangc is required for enhanced shader builds")
        endif()
    elseif(BTB_REQUIRE_VULKAN)
        message(FATAL_ERROR "Enhanced game requires the Vulkan 1.4 SDK")
    else()
        message(WARNING "Vulkan 1.4 SDK absent: enhanced target will only support source smoke tests")
    endif()
endif()
